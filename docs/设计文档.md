#### 2.1.2 核心类设计（实际实现）

```python
# ============================================================
# AI Agent 引擎核心模块
# ============================================================
# 
# 本模块实现了 Plan-and-Execute 模式的 AI Agent
# 执行流程：Plan → Execute → Summarize
#
# 核心组件：
# 1. PlanStep: 执行步骤（含依赖关系）
# 2. ExecutionPlan: 执行计划（多步骤编排）
# 3. PlanExecuteAgent: Agent核心引擎
# 4. AgentService: 业务服务层
#
# Author: CoolAI Team
# Date: 2026-05-13
# ============================================================

from typing import List, Dict, Any, Optional
from dataclasses import dataclass, field
from enum import Enum

class StepStatus(Enum):
    """执行步骤状态"""
    PENDING = "pending"
    RUNNING = "running"
    COMPLETED = "completed"
    FAILED = "failed"

@dataclass
class PlanStep:
    """
    执行步骤
    
    Attributes:
        step_id: 步骤编号
        description: 步骤描述
        tool_name: 使用的工具名称（可选）
        tool_input: 工具输入参数
        depends_on: 依赖的前置步骤ID列表
        status: 步骤执行状态
        result: 步骤执行结果
    """
    step_id: int
    description: str
    tool_name: Optional[str] = None
    tool_input: Optional[Dict[str, Any]] = None
    depends_on: List[int] = field(default_factory=list)
    status: StepStatus = StepStatus.PENDING
    result: Optional[str] = None

@dataclass
class ExecutionPlan:
    """
    执行计划
    
    包含多个步骤，支持步骤间的依赖关系
    """
    steps: List[PlanStep] = field(default_factory=list)
    
    def get_ready_steps(self) -> List[PlanStep]:
        """获取可执行的步骤（依赖已完成）"""
        ready = []
        completed_ids = {s.step_id for s in self.steps if s.status == StepStatus.COMPLETED}
        for step in self.steps:
            if step.status == StepStatus.PENDING:
                if all(dep in completed_ids for dep in step.depends_on):
                    ready.append(step)
        return ready

class PlanExecuteAgent:
    """
    Plan-and-Execute Agent 核心引擎
    
    集成 LLM Gateway、MCP Registry、RAG Pipeline
    支持流式输出(SSE)
    
    三阶段执行：
    1. plan(): LLM生成执行计划
    2. execute(): 逐步执行计划，调用工具
    3. summarize(): 汇总结果生成回答
    """
    
    def __init__(self, llm_gateway, mcp_registry, rag_pipeline=None):
        self.llm_gateway = llm_gateway
        self.mcp_registry = mcp_registry
        self.rag_pipeline = rag_pipeline
    
    async def plan(self, query: str, context: List[Dict] = None) -> ExecutionPlan:
        """LLM生成执行计划"""
        ...
    
    async def execute(self, plan: ExecutionPlan) -> ExecutionPlan:
        """逐步执行计划"""
        ...
    
    async def summarize(self, query: str, plan: ExecutionPlan) -> str:
        """汇总结果生成最终回答"""
        ...

class AgentService:
    """
    Agent业务服务层
    
    对外暴露 chat() 和 chat_stream() 接口
    管理Agent生命周期
    """
    ...
```

### 3.2 核心表结构

> 实际表结构由 SQLAlchemy ORM 自动创建，以下为参考的 SQL 定义

```sql
-- 用户表
CREATE TABLE users (
    id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    username        VARCHAR(50) NOT NULL UNIQUE,
    email           VARCHAR(255) NOT NULL UNIQUE,
    password_hash   VARCHAR(255) NOT NULL,
    role            VARCHAR(20) DEFAULT 'user',
    is_active       BOOLEAN DEFAULT true,
    created_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 对话会话表
CREATE TABLE conversations (
    id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id         UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    title           VARCHAR(255),
    model           VARCHAR(50) DEFAULT 'gpt-4',
    system_prompt   TEXT,
    config          JSONB,
    is_deleted      BOOLEAN DEFAULT false,
    created_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 消息记录表
CREATE TABLE messages (
    id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    conversation_id UUID NOT NULL REFERENCES conversations(id) ON DELETE CASCADE,
    role            VARCHAR(20) NOT NULL,
    content         TEXT NOT NULL,
    metadata        JSONB,
    tokens          INTEGER,
    created_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 知识库表
CREATE TABLE knowledge_bases (
    id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id         UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    name            VARCHAR(255) NOT NULL,
    description     TEXT,
    config          JSONB,
    chunk_count     INTEGER DEFAULT 0,
    doc_count       INTEGER DEFAULT 0,
    created_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 文档表
CREATE TABLE documents (
    id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    knowledge_base_id UUID NOT NULL REFERENCES knowledge_bases(id) ON DELETE CASCADE,
    filename        VARCHAR(255) NOT NULL,
    file_path       VARCHAR(500),
    file_size       BIGINT,
    file_type       VARCHAR(20),
    status          VARCHAR(20) DEFAULT 'pending',
    error_message   TEXT,
    created_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    processed_at    TIMESTAMP
);

-- 文本块表
CREATE TABLE chunks (
    id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    document_id     UUID NOT NULL REFERENCES documents(id) ON DELETE CASCADE,
    content         TEXT NOT NULL,
    chunk_index     INTEGER,
    vector_id       VARCHAR(100),
    metadata        JSONB,
    created_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 声音模型表
CREATE TABLE voice_models (
    id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id         UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    name            VARCHAR(255) NOT NULL,
    description     TEXT,
    model_path      VARCHAR(500),
    reference_audio VARCHAR(500),
    status          VARCHAR(20) DEFAULT 'training',
    created_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- TTS任务表
CREATE TABLE tts_tasks (
    id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id         UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    text            TEXT NOT NULL,
    voice_id        VARCHAR(100),
    status          VARCHAR(20) DEFAULT 'pending',
    output_path     VARCHAR(500),
    error_message   TEXT,
    created_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    completed_at    TIMESTAMP
);

-- 操作日志表
CREATE TABLE operation_logs (
    id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id         UUID REFERENCES users(id),
    action          VARCHAR(100) NOT NULL,
    resource_type   VARCHAR(50),
    resource_id     UUID,
    details         JSONB,
    ip_address      VARCHAR(45),
    user_agent      TEXT,
    created_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 索引
CREATE INDEX idx_messages_conv ON messages(conversation_id);
CREATE INDEX idx_conversations_user ON conversations(user_id);
CREATE INDEX idx_documents_kb ON documents(knowledge_base_id);
CREATE INDEX idx_chunks_doc ON chunks(document_id);
CREATE INDEX idx_tts_tasks_user ON tts_tasks(user_id);
CREATE INDEX idx_logs_user ON operation_logs(user_id);
CREATE INDEX idx_logs_created ON operation_logs(created_at);
```

```yaml
# 流式对话接口
/api/v1/chat/stream:
  post:
    summary: 流式对话(SSE)
    requestBody:
      required: [conversation_id, message]
      properties:
        conversation_id: {type: string}
        message: {type: string}
        model: {type: string, optional}
        use_agent: {type: boolean, optional}      # ⚠️ 前端未对接
        use_knowledge: {type: boolean, optional}
        knowledge_base_id: {type: string, optional}
    responses:
      200:
        content: text/event-stream
        example: |
          data: {"type": "token", "content": "你好"}
          data: {"type": "done", "usage": {"tokens": 100}}

# TTS合成接口
/api/v1/tts/synthesize:
  post:
    summary: 语音合成
    requestBody:
      required: [text]
      properties:
        text: {type: string, maxLength: 10000}
        voice: {type: string}
        speed: {type: number, min: 0.5, max: 2.0}
        format: {type: string, enum: [mp3, wav, ogg]}
    responses:
      200:
        properties:
          success: {type: boolean}
          audio_data: {type: string, description: "Base64编码音频"}
          size: {type: integer}
```
