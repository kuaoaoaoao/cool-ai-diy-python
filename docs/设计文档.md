```python
# ============================================================
# AI Agent 引擎核心模块
# ============================================================
# 
# 本模块实现了基于 LangChain 的 AI Agent 智能代理引擎
# 支持 ReAct (Reasoning + Acting) 推理模式
# 支持 Tool Calling 工具调用
# 支持多种记忆机制（短期会话记忆 + 长期向量记忆）
#
# 主要组件：
# 1. AgentCore: Agent 核心编排器
# 2. ToolRegistry: 工具注册中心
# 3. MemoryManager: 记忆管理器
# 4. LLMRouter: 大模型路由
#
# Author: CoolAI Team
# Date: 2026-05-13
# ============================================================

from typing import List, Dict, Any, Optional
from abc import ABC, abstractmethod
from dataclasses import dataclass, field
from enum import Enum
import asyncio

class AgentType(Enum):
    """
    Agent 类型枚举
    定义了系统中支持的不同类型智能代理
    """
    # 通用对话代理，最基础的问答型代理
    CHAT = "chat"
    # 推理代理，支持 Chain-of-Thought 思维链
    REASONING = "reasoning"
    # 工具代理，支持调用外部工具
    TOOL_CALLING = "tool_calling"
    # 计划执行代理，先规划后执行
    PLAN_EXECUTE = "plan_execute"


@dataclass
class Tool:
    """
    工具定义数据类
    
    Attributes:
        name: 工具名称，唯一标识符
        description: 工具描述，用于LLM理解工具用途
        parameters: 工具参数Schema（JSON Schema格式）
        handler: 工具执行函数
        is_async: 是否异步执行
    """
    name: str                          # 工具名称
    description: str                   # 工具功能描述
    parameters: Dict[str, Any]         # 参数定义
    handler: Any                       # 实际执行函数
    is_async: bool = False            # 是否支持异步调用


@dataclass
class Message:
    """
    对话消息数据类
    
    用于表示Agent与用户的对话消息
    支持role-content结构，兼容OpenAI消息格式
    
    Attributes:
        role: 消息角色 (user/assistant/system/tool)
        content: 消息内容
        metadata: 附加元数据（如工具调用结果等）
    """
    role: str                          # 消息角色
    content: str                       # 消息正文
    metadata: Dict[str, Any] = field(default_factory=dict)  # 扩展信息


class ToolRegistry:
    """
    工具注册中心
    
    负责管理所有可用的AI工具
    提供工具的注册、查找、执行功能
    
    设计模式：注册表模式（Registry Pattern）
    使得工具可以在运行时动态注册和发现
    """
    
    def __init__(self):
        """
        初始化工具注册中心
        
        使用字典存储工具实例，键为工具名称
        """
        # 工具注册表：name -> Tool实例
        self._tools: Dict[str, Tool] = {}
    
    def register(self, tool: Tool) -> None:
        """
        注册一个新工具到注册中心
        
        Args:
            tool: Tool工具实例，包含工具的所有信息
        
        Raises:
            ValueError: 如果工具名称已存在
        """
        # 检查工具名称是否已存在，避免重复注册
        if tool.name in self._tools:
            raise ValueError(f"工具 '{tool.name}' 已存在，请使用其他名称")
        
        # 将工具添加到注册表
        self._tools[tool.name] = tool
    
    def get(self, name: str) -> Optional[Tool]:
        """
        根据名称获取工具实例
        
        Args:
            name: 工具名称
        
        Returns:
            Tool实例，如果不存在则返回None
        """
        return self._tools.get(name)
    
    def list_all(self) -> List[Tool]:
        """
        列出所有已注册的工具
        
        Returns:
            所有Tool实例的列表
        """
        return list(self._tools.values())


class MemoryManager:
    """
    记忆管理器
    
    管理Agent的对话历史和长期记忆
    支持短期会话记忆（基于消息列表）和长期向量记忆（基于RAG）
    
    设计思路：
    - 短期记忆：保存最近N轮对话，支持上下文窗口管理
    - 长期记忆：通过向量数据库存储，支持语义检索
    
    Attributes:
        short_term_limit: 短期记忆保存的最大消息数
        vector_store: 向量数据库连接（用于长期记忆）
    """
    
    def __init__(
        self, 
        short_term_limit: int = 20,
        vector_store: Any = None
    ):
        """
        初始化记忆管理器
        
        Args:
            short_term_limit: 短期记忆的最大消息数
            vector_store: 向量数据库实例（可选）
        """
        # 短期记忆：最近N轮对话
        self.short_term_limit = short_term_limit
        self.short_term: List[Message] = []
        
        # 长期记忆：向量数据库（可选）
        self.vector_store = vector_store
    
    def add_message(self, message: Message) -> None:
        """
        添加一条消息到短期记忆
        
        实现FIFO（先进先出）机制，当超过限制时自动清理旧消息
        
        Args:
            message: 待添加的消息
        """
        self.short_term.append(message)
        
        # 如果超过限制，移除最旧的消息
        if len(self.short_term) > self.short_term_limit:
            self.short_term.pop(0)
    
    def get_context(self) -> List[Message]:
        """
        获取完整的对话上下文
        
        包含短期记忆中的所有消息
        
        Returns:
            消息列表，代表完整的对话历史
        """
        return self.short_term.copy()
    
    def search_long_term(self, query: str, top_k: int = 5) -> List[str]:
        """
        在长期记忆中搜索相关内容
        
        利用向量数据库进行语义相似度检索
        
        Args:
            query: 搜索查询文本
            top_k: 返回的最相似结果数量
        
        Returns:
            相关内容的文本列表
        """
        if self.vector_store is None:
            return []
        
        # 使用向量数据库进行语义搜索
        results = self.vector_store.similarity_search(query, k=top_k)
        return [doc.page_content for doc in results]
    
    def clear(self) -> None:
        """
        清空所有短期记忆
        
        用于开始新的对话会话
        """
        self.short_term.clear()


class LLMRouter:
    """
    大模型路由器
    
    负责统一管理和调度多个大语言模型
    支持模型的热切换、负载均衡、故障转移
    
    支持的模型类型：
    - OpenAI GPT系列
    - Anthropic Claude系列
    - 本地开源模型（ChatGLM、Qwen等）
    
    Attributes:
        models: 已注册的模型字典 {model_name: model_config}
        default_model: 默认使用的模型名称
    """
    
    def __init__(self):
        """初始化模型路由器"""
        # 模型配置字典
        self.models: Dict[str, Dict[str, Any]] = {}
        # 默认模型
        self.default_model: Optional[str] = None
    
    def register_model(
        self,
        name: str,
        api_type: str,
        api_key: str,
        base_url: Optional[str] = None,
        model_id: Optional[str] = None,
        **kwargs
    ) -> None:
        """
        注册一个新的大模型
        
        Args:
            name: 模型标识名称
            api_type: API类型 (openai/anthropic/local/custom)
            api_key: API密钥
            base_url: API基础URL（部分模型需要）
            model_id: 具体模型ID（如gpt-4、claude-3等）
            **kwargs: 其他模型参数
        """
        self.models[name] = {
            "api_type": api_type,
            "api_key": api_key,
            "base_url": base_url,
            "model_id": model_id or name,
            **kwargs
        }
        
        # 如果是第一个注册的模型，设为默认
        if self.default_model is None:
            self.default_model = name
    
    def set_default(self, name: str) -> None:
        """
        设置默认使用的模型
        
        Args:
            name: 模型名称
        
        Raises:
            ValueError: 模型不存在
        """
        if name not in self.models:
            raise ValueError(f"模型 '{name}' 未注册")
        self.default_model = name
    
    def get_model_config(self, name: Optional[str] = None) -> Dict[str, Any]:
        """
        获取指定模型的配置信息
        
        Args:
            name: 模型名称，如果为None则使用默认模型
        
        Returns:
            模型配置字典
        """
        model_name = name or self.default_model
        if model_name not in self.models:
            raise ValueError(f"模型 '{model_name}' 未注册")
        return self.models[model_name]


class AgentCore:
    """
    AI Agent 核心编排器
    
    这是整个Agent系统的核心类，负责：
    1. 协调各个子系统（工具、记忆、模型）
    2. 执行Agent循环（观察-思考-行动）
    3. 管理对话生命周期
    
    设计模式：策略模式 + 模板方法模式
    
    Attributes:
        agent_type: Agent运行类型
        tool_registry: 工具注册中心
        memory_manager: 记忆管理器
        llm_router: 大模型路由器
    """
    
    def __init__(
        self,
        agent_type: AgentType = AgentType.CHAT,
        tool_registry: Optional[ToolRegistry] = None,
        memory_manager: Optional[MemoryManager] = None,
        llm_router: Optional[LLMRouter] = None
    ):
        """
        初始化Agent核心
        
        Args:
            agent_type: Agent类型，决定Agent的行为模式
            tool_registry: 工具注册中心实例
            memory_manager: 记忆管理器实例
            llm_router: 大模型路由器实例
        """
        self.agent_type = agent_type
        # 使用提供的实例或创建新的空实例
        self.tool_registry = tool_registry or ToolRegistry()
        self.memory_manager = memory_manager or MemoryManager()
        self.llm_router = llm_router or LLMRouter()
        
        # Agent运行时状态
        self._is_running = False
        self._max_iterations = 10  # 最大迭代次数，防止无限循环
    
    def add_tool(self, tool: Tool) -> None:
        """
        向Agent添加工具
        
        Args:
            tool: 待添加的工具实例
        """
        self.tool_registry.register(tool)
    
    async def chat(
        self,
        user_input: str,
        model_name: Optional[str] = None,
        stream: bool = True,
        **kwargs
    ) -> Dict[str, Any]:
        """
        处理用户对话输入
        
        这是Agent的主入口方法，处理单轮对话请求
        
        Args:
            user_input: 用户输入的文本
            model_name: 使用的模型名称（可选）
            stream: 是否使用流式输出
            **kwargs: 其他参数传递给模型
        
        Returns:
            包含回复内容的字典
        """
        # 构建用户消息
        user_message = Message(role="user", content=user_input)
        # 添加到记忆
        self.memory_manager.add_message(user_message)
        
        # 根据Agent类型选择处理策略
        if self.agent_type == AgentType.CHAT:
            return await self._chat_mode(model_name, stream, **kwargs)
        elif self.agent_type == AgentType.TOOL_CALLING:
            return await self._tool_calling_mode(model_name, stream, **kwargs)
        elif self.agent_type == AgentType.REASONING:
            return await self._reasoning_mode(model_name, stream, **kwargs)
        else:
            return await self._chat_mode(model_name, stream, **kwargs)
    
    async def _chat_mode(
        self,
        model_name: Optional[str],
        stream: bool,
        **kwargs
    ) -> Dict[str, Any]:
        """
        基础聊天模式
        
        最简单的Agent模式，直接将对话历史发送给LLM
        
        Args:
            model_name: 模型名称
            stream: 是否流式输出
            **kwargs: 其他参数
        
        Returns:
            包含回复的字典
        """
        # 获取模型配置
        model_config = self.llm_router.get_model_config(model_name)
        
        # 构建对话上下文
        context = self.memory_manager.get_context()
        
        # TODO: 调用实际的LLM API
        # 这里预留接口，后续会接入具体的LLM调用逻辑
        
        response_content = f"[模拟回复] 收到了: {context[-1].content if context else user_input}"
        
        # 构建助手消息
        assistant_message = Message(role="assistant", content=response_content)
        self.memory_manager.add_message(assistant_message)
        
        return {
            "content": response_content,
            "model": model_config.get("model_id", "unknown"),
            "usage": {"tokens": len(response_content)}
        }
    
    async def _tool_calling_mode(
        self,
        model_name: Optional[str],
        stream: bool,
        **kwargs
    ) -> Dict[str, Any]:
        """
        工具调用模式
        
        支持ReAct风格的工具调用循环：
        1. Thought: LLM思考下一步应该做什么
        2. Action: 调用相应工具
        3. Observation: 获取工具执行结果
        4. 重复直到得到最终答案
        
        Args:
            model_name: 模型名称
            stream: 是否流式输出
            **kwargs: 其他参数
        
        Returns:
            包含回复和工具调用历史的字典
        """
        # 获取上下文和可用的工具列表
        context = self.memory_manager.get_context()
        available_tools = self.tool_registry.list_all()
        
        # 构建工具描述（供LLM理解可用的工具）
        tool_descriptions = [
            {
                "name": tool.name,
                "description": tool.description,
                "parameters": tool.parameters
            }
            for tool in available_tools
        ]
        
        # TODO: 实现完整的ReAct循环
        # 当前为占位实现
        
        return {
            "content": "Tool calling mode - 待实现",
            "tool_calls": [],
```

```python
class TextChunker:
    """
    文本分块器
    
    负责将长文档切分成适合检索的小块
    支持多种分块策略
    
    设计考量：
    1. 块大小：太大会引入噪声，太小会丢失上下文
    2. 块重叠：增加重叠区域可以提高召回率
    3. 保留边界：尽量在语义完整处切分
    """
    
    def __init__(
        self,
        strategy: ChunkStrategy = ChunkStrategy.FIXED_SIZE,
        chunk_size: int = 500,
        chunk_overlap: int = 50
    ):
        """
        初始化分块器
        
        Args:
            strategy: 分块策略
            chunk_size: 目标块大小（字符数或token数）
            chunk_overlap: 相邻块的重叠大小
        """
        self.strategy = strategy
        self.chunk_size = chunk_size
        self.chunk_overlap = chunk_overlap
    
    def chunk_document(self, document: Document) -> List[TextChunk]:
        """
        将文档分块
        
        根据配置的策略将文档切分成多个文本块
        
        Args:
            document: 待分块的文档
        
        Returns:
            TextChunk列表
        """
        if self.strategy == ChunkStrategy.FIXED_SIZE:
            return self._fixed_size_chunk(document)
        elif self.strategy == ChunkStrategy.RECURSIVE:
            return self._recursive_chunk(document)
        elif self.strategy == ChunkStrategy.PARAGRAPH:
            return self._paragraph_chunk(document)
        else:
            return self._fixed_size_chunk(document)
    
    def _fixed_size_chunk(self, document: Document) -> List[TextChunk]:
        """
        固定长度分块
        
        按固定字符数切分文档，简单高效
        适合内容结构不明显的文档
        
        Args:
            document: 待分块文档
        
        Returns:
            文本块列表
        """
        content = document.content
        chunks = []
        
        # 滑动窗口切分
        start = 0
        index = 0
        
        while start < len(content):
            # 计算当前块的结束位置
            end = start + self.chunk_size
            
            # 尽量在句子边界处切分
            if end < len(content):
                # 向前查找最近的句号或换行符
                for i in range(min(end, len(content) - 1), max(start, end - 100), -1):
                    if content[i] in '。！？.\n':
                        end = i + 1
                        break
            
            # 提取块内容
            chunk_content = content[start:end].strip()
            
            if chunk_content:  # 跳过空块
                chunks.append(TextChunk(
                    chunk_id=f"{document.doc_id}_{index}",
                    content=chunk_content,
                    doc_id=document.doc_id,
                    chunk_index=index,
                    metadata={
                        "source": document.metadata.get("source"),
                        "title": document.title
                    }
                ))
            
            # 移动窗口（考虑重叠）
            start = end - self.chunk_overlap if end < len(content) else len(content)
            index += 1
        
        return chunks
    
    def _recursive_chunk(self, document: Document) -> List[TextChunk]:
        """
        递归分块
        
        从大到小依次尝试切分：
        段落 -> 句子 -> 固定长度
        
        尽量保持语义完整性
        
        Args:
            document: 待分块文档
        
        Returns:
            文本块列表
        """
        # TODO: 实现递归分块逻辑
        # 1. 先按段落分割
        # 2. 如果段落过长，再按句子分割
        # 3. 如果句子仍过长，再按固定长度分割
        return self._fixed_size_chunk(document)
    
    def _paragraph_chunk(self, document: Document) -> List[TextChunk]:
        """
        段落分块
        
        按自然段落分割文档
        适合结构清晰的文档（如文章、报告）
        
        Args:
            document: 待分块文档
        
        Returns:
            文本块列表
        """
        content = document.content
        # 按换行符分割段落
        paragraphs = [p.strip() for p in content.split('\n\n') if p.strip()]
        
        chunks = []
        for index, para in enumerate(paragraphs):
            # 过滤过短的段落
            if len(para) < 20:
                continue
            
            chunks.append(TextChunk(
                chunk_id=f"{document.doc_id}_{index}",
                content=para,
                doc_id=document.doc_id,
                chunk_index=index,
                metadata={
                    "source": document.metadata.get("source"),
                    "title": document.title,
                    "paragraph_num": index + 1
                }
            ))
        
        return chunks

class RRFusion:
    """
    RRF (Reciprocal Rank Fusion) 排名融合
    
    将多个检索结果进行融合排序
    RRF是一种简单而有效的融合方法：
    score = Σ(1 / (k + rank_i))
    其中k通常取60
    
    优势：
    - 不需要训练
    - 对不同规模的检索结果都好
    - 简单易实现

    def __init__(self, k: int = 60):
        """
        初始化RRF融合器
        
        Args:
            k: RRF公式中的常数，用于平滑排名差异
        """
        self.k = k
    
    def fuse(

        result_lists: List[List[SearchResult]]
    ) -> List[SearchResult]:
        """
        融合多个检索结果
        
        Args:
            result_lists: 多个检索结果列表
        
        Returns:
            融合后的排序结果
        """
        # 合并所有结果的得分
        chunk_scores: Dict[str, float] = {}
        
        for result_list in result_lists:
            for rank, result in enumerate(result_list, start=1):
                chunk_id = result.chunk.chunk_id

                if chunk_id in chunk_scores:

        初始化RAG Pipeline
        
        Args:
            embedding_model: 嵌入模型实例
            vector_store: 向量存储实例
            keyword_searcher: 关键词搜索实例
            llm_router: LLM路由器（用于生成）
        """
        self.embedding_model = embedding_model or EmbeddingModel()
        self.vector_store = vector_store or MilvusVectorStore()
        self.keyword_searcher = keyword_searcher
        self.llm_router = llm_router
        
        # 创建默认的分块器
        self.chunker = TextChunker(
            strategy=ChunkStrategy.PARAGRAPH,
            chunk_size=500,
            chunk_overlap=50
        )
        
        # 创建RRF融合器
        self.rrf_fusion = RRFusion(k=60)
    
    def add_document(
        self,
        file_path: str,
        parser: Optional[DocumentParser] = None
    ) -> Dict[str, Any]:
        """
        添加文档到知识库
        
        完整流程：解析 -> 分块 -> 向量化 -> 存储
        
        Args:
            file_path: 文档路径
            parser: 文档解析器（可选，自动推断）
        
        Returns:
            操作结果字典
        """
        # 1. 选择合适的解析器
        if parser is None:
            parser = self._auto_select_parser(file_path)
        
        if parser is None:
            return {"success": False, "error": "不支持的文档格式"}
        
        # 2. 解析文档
        document = parser.parse(file_path)
        
        # 3. 分块
        chunks = self.chunker.chunk_document(document)
        
        # 4. 向量化
        texts = [chunk.content for chunk in chunks]
        embeddings = self.embedding_model.encode(texts)
        
        # 5. 存储到向量数据库
        # 同时保存chunk_id到vector_id的映射
        for i, chunk in enumerate(chunks):
            chunk.vector_id = f"vec_{chunk.chunk_id}"
        
        success = self.vector_store.insert(chunks, embeddings)
        
        return {
            "success": success,
            "doc_id": document.doc_id,
            "title": document.title,
            "chunk_count": len(chunks)
        }
    
    def _auto_select_parser(self, file_path: str) -> Optional[DocumentParser]:
        """
        自动选择合适的文档解析器
        
        根据文件扩展名选择解析器
        
        Args:
            file_path: 文件路径
        
        Returns:
            合适的解析器，或None
        """
        file_path_lower = file_path.lower()
        
        if file_path_lower.endswith('.txt'):
            return TxtParser()
        elif file_path_lower.endswith('.pdf'):
            return PDFParser()
        # TODO: 添加更多解析器
        else:
            return None
    
    async def retrieve(
        self,
        query: str,
        top_k: int = 5,
        use_hybrid: bool = True
    ) -> List[SearchResult]:
        """
        检索相关文档块
        
        支持纯向量检索或混合检索（向量 + 关键词）
        
        Args:
            query: 用户查询
            top_k: 返回结果数量
            use_hybrid: 是否使用混合检索
        
        Returns:
            检索结果列表
        """
        # 1. 将查询向量化
        query_vector = self.embedding_model.encode_single(query)
        
        # 2. 向量检索
        vector_results = self.vector_store.search(query_vector, top_k)
        
        if not use_hybrid or self.keyword_searcher is None:
            return vector_results
        
        # 3. 关键词检索
        keyword_results = self.keyword_searcher.search(query, top_k)
        
        # 4. RRF融合
        # 需要转换为SearchResult格式
        # 暂时返回向量检索结果
        return vector_results
    
    def build_prompt(
        self,
        query: str,
        context_chunks: List[SearchResult]
    ) -> str:
        """
        构建增强提示词
        
        将检索到的上下文注入到Prompt中
        
        Prompt模板：
        你是一个助手，请根据以下参考资料回答问题。
        
        参考资料：
        {context}
        
        问题：{query}
        
        回答：
        
        Args:
            query: 用户问题
            context_chunks: 检索到的相关文档块
        
        Returns:
            构建好的提示词
        """
        # 拼接上下文
        context_parts = []
        for i, result in enumerate(context_chunks, start=1):
            source = result.chunk.metadata.get('source', '未知来源')
            context_parts.append(
                f"[参考资料{i}](来源: {source})\n{result.chunk.content}"
            )
        
        context_text = "\n\n".join(context_parts)
        
        # 构建完整Prompt
        prompt = f"""请根据以下参考资料回答问题。如果资料中没有相关信息，请如实说明。

参考资料：
{context_text}

问题：{query}

回答："""
        
        return prompt
    
    async def query(
        self,
        user_query: str,
        top_k: int = 5,
        use_knowledge: bool = True,
        model_name: Optional[str] = None
    ) -> Dict[str, Any]:
        """
        RAG查询主入口
        
        执行完整的检索-增强-生成流程
        
        Args:
            user_query: 用户问题
            top_k: 检索结果数量
            use_knowledge: 是否使用知识库
            model_name: 使用的模型名称
        
        Returns:
            包含答案和检索信息的字典
        """
        if use_knowledge:
            # 1. 检索相关文档
            retrieval_results = await self.retrieve(user_query, top_k)
            
            # 2. 构建增强Prompt
            prompt = self.build_prompt(user_query, retrieval_results)
            
            # 3. 调用LLM生成答案
            # TODO: 实际调用LLM
            answer = f"[模拟回答] 基于{len(retrieval_results)}个参考资料回答"
            
            return {
                "answer": answer,
                "prompt": prompt,
                "retrieval_results": [
                    {
                        "content": r.chunk.content,
                        "score": r.score,
                        "source": r.chunk.metadata.get('source', '')
                    }
                    for r in retrieval_results
                ],
                "model": model_name or "default"
            }
        else:
            # 不使用知识库的直接问答
            return {
                "answer": f"[模拟回答] {user_query}",
                "retrieval_results": [],
                "model": model_name or "default"
            }
```

```python
    """
    声音克隆引擎

    使用克隆的声音进行语音合成

    1. 用户上传参考音频（30秒以上）
    2. 系统提取声纹特征
    3. 使用克隆的声音进行TTS合成
    
    Attributes:
        voice_models: 已克隆的声音模型字典

        """
        初始化声音克隆引擎

            model_dir: 克隆声音模型存储目录

        self.model_dir = Path(model_dir)
        self.model_dir.mkdir(parents=True, exist_ok=True)
        
        # 已克隆的声音模型
        self.voice_models: Dict[str, Dict[str, Any]] = {}
        
        # 加载已存在的克隆声音
        self._load_existing_models()
    
    def _load_existing_models(self):
        """
        加载已存在的克隆声音模型
        
        扫描模型目录，加载所有克隆的声音信息

        if not self.model_dir.exists():
            return
        
        for voice_dir in self.model_dir.iterdir():
            if voice_dir.is_dir():
                config_file = voice_dir / "config.json"
                if config_file.exists():
                    import json
                    with open(config_file) as f:
                        config = json.load(f)
                    self.voice_models[voice_dir.name] = config

    负责：
    - 引擎选择和配置
    - 任务队列管理
    - 结果存储
    - API限流
    
    Attributes:
        engines: 已注册的TTS引擎
        default_engine: 默认引擎
        task_queue: TTS任务队列
    """
    
    def __init__(self):
        """
        初始化TTS服务
        """
        # 引擎字典
        self.engines: Dict[TTSEngineType, TTSEngineBase] = {}
        self.default_engine: TTSEngineType = TTSEngineType.EDGE_TTS
        
        # 初始化内置引擎
        self._register_default_engines()
    
    def _register_default_engines(self):
        """
        注册默认的TTS引擎
        
        包括Edge-TTS和声音克隆引擎
        """
        # 注册Edge-TTS
        self.register_engine(TTSEngineType.EDGE_TTS, EdgeTTSEngine())
        
        # 注册声音克隆引擎
        self.register_engine(TTSEngineType.VOICE_CLONE, VoiceCloneEngine())
    
    def register_engine(
        self,
        engine_type: TTSEngineType,
        engine: TTSEngineBase
    ):
        """
        注册TTS引擎
        
        Args:
            engine_type: 引擎类型
            engine: 引擎实例
        """
        self.engines[engine_type] = engine
        
        # 如果是第一个注册的引擎，设为默认
        if len(self.engines) == 1:
            self.default_engine = engine_type
    
    def get_engine(self, engine_type: Optional[TTSEngineType] = None) -> TTSEngineBase:
        """
        获取TTS引擎
        
        Args:
            engine_type: 引擎类型，None则使用默认
        
        Returns:
            TTS引擎实例
        
        Raises:
            ValueError: 引擎不存在
        """
        if engine_type is None:
            engine_type = self.default_engine
        
        if engine_type not in self.engines:
            raise ValueError(f"TTS引擎不存在: {engine_type}")
        
        return self.engines[engine_type]
    
    def list_all_voices(self) -> List[VoiceInfo]:
        """
        列出所有可用的语音
        
        聚合所有引擎的语音
        
        Returns:
            VoiceInfo列表
        """
        all_voices = []
        seen_ids = set()
        
        for engine in self.engines.values():
            for voice in engine.list_voices():
                if voice.voice_id not in seen_ids:
                    all_voices.append(voice)
                    seen_ids.add(voice.voice_id)
        
        return all_voices
    
    async def synthesize(
        self,
        text: str,
        output_path: Optional[str] = None,
        config: Optional[TTSConfig] = None
    ) -> Dict[str, Any]:
        """
        合成语音
        
        主入口方法，处理语音合成请求
        
        Args:
            text: 待合成的文本
            output_path: 输出文件路径（可选）
            config: 合成配置（可选）
        
        Returns:
            包含音频数据或文件路径的字典
        """
        # 使用默认配置
        if config is None:
            config = TTSConfig()
        
        # 获取引擎
        engine = self.get_engine(config.engine)
        
        # 合成音频
        audio_data = await engine.synthesize(text, config)
        
        # 如果指定了输出路径，保存文件
        if output_path:
            Path(output_path).parent.mkdir(parents=True, exist_ok=True)
            with open(output_path, 'wb') as f:
                f.write(audio_data)
            
            return {
                "success": True,
                "output_path": output_path,
                "size": len(audio_data)
            }
        
        # 返回Base64编码的音频
        audio_base64 = base64.b64encode(audio_data).decode('utf-8')
        
        return {
            "success": True,
            "audio_data": f"data:audio/{config.format.value};base64,{audio_base64}",
            "size": len(audio_data)
        }
    
    async def synthesize_stream(
        self,
        text: str,
        config: Optional[TTSConfig] = None
    ) -> AsyncIterator[Dict[str, Any]]:
        """
        流式合成语音
        
        适用于长文本的实时合成
        
        Args:
            text: 待合成的文本
            config: 合成配置
        
        Yields:
            音频数据块字典
        """
        if config is None:
            config = TTSConfig()
        
        engine = self.get_engine(config.engine)
        
        async for chunk in engine.synthesize_stream(text, config):
            yield {
                "type": "audio",
                "data": base64.b64encode(chunk).decode('utf-8')
            }
    
    async def batch_synthesize(
        self,
        tasks: List[Dict[str, Any]]
    ) -> List[Dict[str, Any]]:
        """
        批量合成语音
        
        并行处理多个合成任务
        
        Args:
            tasks: 任务列表，每个任务包含text、output_path等
        
        Returns:
            结果列表
        """
        results = await asyncio.gather(
            *[self.synthesize(**task) for task in tasks],
            return_exceptions=True
        )
        
        return [
            r if not isinstance(r, Exception) else {"success": False, "error": str(r)}
            for r in results
        ]
```

```sql
-- ============================================================
-- 用户表
-- ============================================================
CREATE TABLE users (
    id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    username        VARCHAR(50) NOT NULL UNIQUE,
    email           VARCHAR(255) NOT NULL UNIQUE,
    password_hash   VARCHAR(255) NOT NULL,
    role            VARCHAR(20) DEFAULT 'user',  -- admin/user
    is_active       BOOLEAN DEFAULT true,
    created_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- ============================================================
-- 对话会话表
-- ============================================================
CREATE TABLE conversations (
    id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id         UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    title           VARCHAR(255),
    model           VARCHAR(50) DEFAULT 'gpt-4',
    system_prompt   TEXT,
    config          JSONB,  -- 存储LLM参数配置
    is_deleted      BOOLEAN DEFAULT false,
    created_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- ============================================================
-- 消息记录表
-- ============================================================
CREATE TABLE messages (
    id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    conversation_id UUID NOT NULL REFERENCES conversations(id) ON DELETE CASCADE,
    role            VARCHAR(20) NOT NULL,  -- user/assistant/system/tool
    content         TEXT NOT NULL,
    metadata        JSONB,  -- 工具调用结果等
    tokens          INTEGER,
    created_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- ============================================================
-- 知识库表
-- ============================================================
CREATE TABLE knowledge_bases (
    id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id         UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    name            VARCHAR(255) NOT NULL,
    description     TEXT,
    config          JSONB,  -- 分块策略、检索参数等
    chunk_count     INTEGER DEFAULT 0,
    doc_count       INTEGER DEFAULT 0,
    created_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- ============================================================
-- 文档表
-- ============================================================
CREATE TABLE documents (
    id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    knowledge_base_id UUID NOT NULL REFERENCES knowledge_bases(id) ON DELETE CASCADE,
    filename        VARCHAR(255) NOT NULL,
    file_path       VARCHAR(500),
    file_size       BIGINT,
    file_type       VARCHAR(20),
    status          VARCHAR(20) DEFAULT 'pending',  -- pending/processing/completed/failed
    error_message   TEXT,
    created_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    processed_at    TIMESTAMP
);

-- ============================================================
-- 文本块表（可选，部分数据存在向量数据库）
-- ============================================================
CREATE TABLE chunks (
    id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    document_id     UUID NOT NULL REFERENCES documents(id) ON DELETE CASCADE,
    content         TEXT NOT NULL,
    chunk_index     INTEGER,
    vector_id       VARCHAR(100),  -- 向量数据库中的ID
    metadata        JSONB,
    created_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- ============================================================
-- 声音模型表
-- ============================================================
CREATE TABLE voice_models (
    id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id         UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    name            VARCHAR(255) NOT NULL,
    description     TEXT,
    model_path      VARCHAR(500),
    reference_audio VARCHAR(500),
    status          VARCHAR(20) DEFAULT 'training',  -- training/ready/failed
    created_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- ============================================================
-- TTS任务表
-- ============================================================
CREATE TABLE tts_tasks (
    id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id         UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    text            TEXT NOT NULL,
    voice_id        VARCHAR(100),
    status          VARCHAR(20) DEFAULT 'pending',
    output_path     VARCHAR(500),
    error_message   TEXT,
    created_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    completed_at    TIMESTAMP
);

-- ============================================================
-- 操作日志表
-- ============================================================
CREATE TABLE operation_logs (
    id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id         UUID REFERENCES users(id),
    action          VARCHAR(100) NOT NULL,
    resource_type   VARCHAR(50),
    resource_id     UUID,
    details         JSONB,
    ip_address      VARCHAR(45),
    user_agent      TEXT,
    created_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 索引
CREATE INDEX idx_messages_conv ON messages(conversation_id);
CREATE INDEX idx_conversations_user ON conversations(user_id);
CREATE INDEX idx_documents_kb ON documents(knowledge_base_id);
CREATE INDEX idx_chunks_doc ON chunks(document_id);
CREATE INDEX idx_tts_tasks_user ON tts_tasks(user_id);
CREATE INDEX idx_logs_user ON operation_logs(user_id);
CREATE INDEX idx_logs_created ON operation_logs(created_at);
```
